SPICETIFY

# 1. Créez le dossier que Spicetify cherche désespérément
mkdir -p ~/.config/spicetify/Themes/marketplace

# 2. Configurez le thème sur 'marketplace' comme demandé par Spotify
spicetify config current_theme marketplace

# 3. Appliquez les changements
spicetify apply
